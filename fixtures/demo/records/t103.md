The captured routine removes one leading version marker and validates the number of fields before converting each field.

def release_key(label):
    parts = label.removeprefix("v").split(".")
    if len(parts) != 3:
        raise ValueError("expected three release fields")
    return tuple(int(part) for part in parts)
